% labelindent+labelwidth+labelsep = leftmargin puts the [n] label exactly on the left
% text margin. The option list is built as a macro because enumitem's keyval parser
% splits on commas before expansion, so a bare \ifcvresume inside [...] silently
% swallows the keys that follow it.
% enumitem does not expand a macro passed as the whole option list, so both variants
% are written out literally.

  \newcommand{\resumePubItemListStart}{%

      \setcounter{pubCounter}{0}%
  }
\else
  \newcommand{\resumePubItemListStart}{%
    \begin{itemize}[leftmargin=18pt, label={}, rightmargin=0pt, parsep=3pt]%
      \setcounter{pubCounter}{0}%
  }
\fi

  \end{itemize}%
  \vspace{-9.1pt}%

% ---- variant tagging -------------------------------------------------
% \tagged{tag,tag}{content} emits content only when \cvvariant matches one
% of the tags. \cvvariant is \def'd in each variant file before \input{preamble};
% the \providecommand below is only a fallback. \cvvariant=full matches everything.
\usepackage{etoolbox}
\providecommand{\cvvariant}{full}
\newtoggle{cvmatch}
\newcommand{\cvchecktag}[1]{\ifdefstring{\cvvariant}{#1}{\toggletrue{cvmatch}}{}}
\newcommand{\tagged}[2]{%
  \togglefalse{cvmatch}%
  \ifdefstring{\cvvariant}{full}{\toggletrue{cvmatch}}{\forcsvlist{\cvchecktag}{#1}}%
  \iftoggle{cvmatch}{#2}{}%
}
